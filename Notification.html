document.getElementById('continueButton').addEventListener('click', function() { const formWrapper = document.querySelector('.eligibility'); const eligibilitySection = document.querySelector('.eligibility-Sect'); const nonEligibilitySection = document.querySelector('.noneligibility-Sect'); // Hide both sections initially eligibilitySection.style.display = 'none'; nonEligibilitySection.style.display = 'none'; // Check which button was clicked and show the appropriate section if (document.querySelector('#fiveyes').checked) { eligibilitySection.style.display = 'block'; // Show eligibility section } else if (document.querySelector('#fiveNo').checked) { nonEligibilitySection.style.display = 'block'; // Show non-eligibility section } // Hide the eligible form-wrapper div formWrapper.style.display = 'none'; });
